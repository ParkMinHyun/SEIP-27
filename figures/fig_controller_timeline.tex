% Two-capture execution contract for the Budget-Aware Draft Controller.
%
% NO NOTATION IN THIS DRAWING, DELIBERATELY.  It is the overview figure and it
% appears before Section III-B defines the model and Section III-D defines the
% pacing quantities, so every label here is prose.  An earlier revision anchored
% the symbols of Eq. (3) on the drawing; that tied figure to equation nicely but
% forward-referenced notation the reader had not met yet.  If a quantity needs a
% symbol, it belongs in Section III-D, not here.
%
% DRAWN AT FINAL SIZE.  Do not wrap this in \resizebox: every label size is
% chosen to sit between \tiny and \scriptsize against the body text, and
% scaling the box breaks that relation.  Natural width is ~17.3cm including the
% lane labels, against \textwidth ~18.1cm in IEEEtran conference mode.
%
% TWO RULES KEEP THIS DRAWING FROM COLLIDING WITH ITSELF.  An earlier revision
% broke both and every box in the Draft lane overlapped its neighbour.
%
%   1. Every box in the Draft lane carries `minimum width`, and the centres
%      below are spaced by that width plus a gap.  Never let a Draft-lane box
%      size itself from its text: one wider label then pushes into the next.
%      Prose labels are wider than symbols were, which is why the boxes are on
%      two lines and the Draft blocks are wider than in the notation version.
%   2. Every multi-line box breaks its lines with an explicit \\ and declares
%      NO `text width`.  Automatic wrapping is font-metric dependent, so the
%      same source gave three lines here and four elsewhere, and the fourth
%      line dropped out of the band into the channel below it.
%
% Horizontal positions show event order and relative occupancy, not measured
% duration.  The drawing is still consistent with a positive delay: the queued
% work plus the two-Draft reserve reach past the deadline guide.
%
% Vertical geometry.  The two guides span the whole height so the measure rows
% at the bottom can reference the deadline drawn on the top lane.  Do not move
% a band without the channel beside it:
%   app band     3.40 .. 4.20     lane channel  2.95 .. 3.40
%   ctrl band    1.90 .. 2.95     skip channel  1.20 .. 1.90  (sticky arc)
%   worker band  0.10 .. 1.20     measure rows  0.02 .. -0.64
%
% NOTE: 'cap' is a reserved TikZ key (line cap); the commit tick is 'commit'.
\begin{tikzpicture}[
  x=1cm, y=1cm, >=latex, font=\scriptsize,
  lanelab/.style={anchor=east, align=right, font=\scriptsize\bfseries,
    text=black!65},
  band/.style={rounded corners=2.5pt, draw=none},
  evt/.style={draw=black!45, fill=white, rounded corners=1.5pt,
    align=center, inner sep=2.5pt, minimum height=0.44cm},
  ctrl/.style={draw=orange!72!black, fill=orange!9, rounded corners=2pt,
    align=center, inner sep=3pt, semithick, minimum height=0.95cm},
  hinge/.style={draw=orange!72!black, fill=orange!35, circle, inner sep=0pt,
    minimum size=0.17cm},
  dbox/.style={draw=blue!52!black, fill=blue!5, rounded corners=2.5pt,
    semithick},
  nbox/.style={draw=blue!52!black, fill=white, rounded corners=1.5pt,
    align=center, inner sep=1.5pt, minimum height=0.52cm, minimum width=1.60cm},
  sbox/.style={draw=purple!62!black, fill=purple!8, rounded corners=1.5pt,
    dash pattern=on 2pt off 1.3pt, align=center, inner sep=1.5pt,
    minimum height=0.52cm, minimum width=1.60cm, text=purple!55!black},
  flow/.style={->, semithick, black!45},
  sticky/.style={->, semithick, densely dashed, purple!62!black},
  commit/.style={black!75, line width=1pt},
  meas/.style={<->, semithick, black!60},
  tag/.style={font=\tiny, text=black!60, inner sep=1.2pt},
  ontop/.style={fill=white, inner sep=1.2pt}
]

  %% ---------------------------------------------------------------- lanes
  \fill[band, black!4] (0.10,3.40) rectangle (15.90,4.20);
  \fill[band, black!4] (0.10,1.90) rectangle (15.90,2.95);
  \fill[band, black!4] (0.10,0.10) rectangle (15.90,1.20);
  \node[lanelab] at (0.00,3.80) {App-facing\\capture};
  \node[lanelab] at (0.00,2.42) {Controller};
  \node[lanelab] at (0.00,0.65) {Draft worker};

  %% --------------------------------------------------- full-height guides
  % The decision guide is also the link from the decision box up to the start
  % of the delay span, which is why neither carries its own arrow.
  \draw[black!35, densely dotted] (2.20,-0.76) -- (2.20,4.32);
  \node[tag, anchor=south] at (2.20,4.32) {decision instant};
  % x=12.63 is the gap between the next Draft's skipped and Encoding boxes, so
  % the guide crosses the worker lane in clear space rather than behind a box
  % fill.  It must stay equal to the right end of the deadline measure below.
  \draw[red!62!black, dashed, semithick] (12.63,-0.76) -- (12.63,4.26);
  \node[tag, anchor=south, text=red!62!black] at (12.63,4.26)
    {this capture's deadline};

  %% ------------------------------------------------------- app-facing lane
  \draw[commit] (0.50,3.36) -- (0.50,4.24);
  \node[tag, anchor=south, align=center, text=black!75]
    at (0.50,4.26) {this capture\\committed};
  \node[evt] (hal) at (1.42,3.80) {HAL ready};

  % The delay is the one user-visible cost; give it a filled span, not a rule.
  \fill[orange!22, rounded corners=1pt] (2.20,3.63) rectangle (4.30,3.97);
  \draw[orange!72!black, semithick, rounded corners=1pt]
    (2.20,3.63) rectangle (4.30,3.97);
  \node[font=\scriptsize, text=orange!42!black] at (3.25,3.80) {pacing delay};

  % `rel' sizes itself from \texttt{captureAvailable} (~2.5cm wide), so leave
  % half a centimetre on each side or its two arrows collapse to arrowheads.
  \node[evt] (rel) at (5.80,3.80) {release\\\texttt{captureAvailable}};
  \draw[flow] (4.30,3.80) -- (rel.west);
  \draw[flow, densely dotted] (rel.east) -- (7.26,3.80);
  \draw[commit] (7.40,3.36) -- (7.40,4.24);
  \node[tag, anchor=south, align=center, text=black!75]
    at (7.40,4.26) {next capture\\committed};

  %% ----------------------------------------------------------- controller
  \node[ctrl] (dec) at (2.20,2.42)
    {\textbf{pacing decision}\\
     weigh queued work and\\
     reserve against time left};
  \draw[flow] (hal.south) -- (1.42,2.95);

  \node[ctrl] (drain) at (14.80,2.42) {queue drain\\clears burst state};

  %% --------------------------------------------------------- Draft worker
  % Everything still admitted when the decision is taken is the backlog.
  \draw[pattern=north east lines, pattern color=black!20, draw=black!30]
    (0.50,0.15) rectangle (3.40,1.15);
  \node[ontop, tag] at (1.32,0.95) {earlier Drafts};
  % No measure arrow here: the span is only 1.2cm and its own label leaves no
  % room for the two arrowheads.  The decision guide marks where the backlog
  % starts and the Draft block marks where it ends, so a plain tag is enough.
  \node[ontop, tag] at (2.80,0.52) {still queued};

  % This Draft: a model rejection skips the single-frame stage for the burst.
  \draw[dbox] (3.40,0.15) rectangle (8.44,1.15);
  \node[anchor=west, font=\scriptsize\bfseries, text=blue!45!black]
    at (3.50,0.97) {this Draft};
  \node[nbox] at (4.37,0.52) {multi-frame\\admitted};
  \node[sbox] (skip) at (6.12,0.52) {single-frame\\skipped};
  \node[nbox, minimum width=1.20cm] at (7.67,0.52) {Encoding\\mandatory};

  \draw[pattern=north east lines, pattern color=black!20, draw=black!30]
    (8.44,0.15) rectangle (9.04,1.15);

  % Next Draft: the same skip, now forced by the burst state, not re-tested.
  \draw[dbox] (9.04,0.15) rectangle (14.08,1.15);
  \node[anchor=west, font=\scriptsize\bfseries, text=blue!45!black]
    at (9.14,0.97) {next Draft};
  \node[nbox] at (10.01,0.52) {multi-frame\\admitted};
  \node[sbox] (forced) at (11.76,0.52) {single-frame\\skipped};
  \node[nbox, minimum width=1.20cm] at (13.31,0.52) {Encoding\\mandatory};

  \draw[flow] (14.08,0.65) -| (drain.south);

  %% --------------------------- FIFO hinges: one Draft start per decision
  \node[hinge] at (3.40,1.15) {};
  \node[hinge] at (9.04,1.15) {};
  % Placed to the RIGHT of the first hinge.  To its left the caption would have
  % to straddle the decision guide, which then prints through the text.
  \node[tag, anchor=south west, align=left] at (3.54,1.31)
    {each Draft start consumes one pending\\decision and refreshes the reserve};

  %% --------------------- the coupling: a skip reshapes the later reserve
  % Shallow out/in angles on purpose.  The arc spans ~5.6cm, so anything past
  % ~40 degrees puts its apex through the Controller band above.
  \draw[sticky] (skip.north) to[out=35, in=145] (forced.north);
  \node[ontop, font=\tiny, text=purple!55!black, anchor=south]
    at (8.94,1.60) {the skip sticks and shrinks the next reserve};

  %% ---------------------------- what the two priced spans cover, in prose
  \draw[decorate, decoration={brace, mirror, amplitude=4pt}, black!55]
    (3.40,0.02) -- (14.08,0.02);
  \node[anchor=north, font=\scriptsize] at (8.74,-0.14)
    {reserve for this Draft and the one the delay releases};

  \draw[meas] (2.20,-0.64) -- (12.63,-0.64);
  \node[ontop, font=\scriptsize] at (7.42,-0.64)
    {time left on the newest committed deadline};

\end{tikzpicture}
