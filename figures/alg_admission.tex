\begin{algorithm}[t]
\caption{Remaining-Sequence Admission}
\label{alg:admission}
\footnotesize
\renewcommand{\algorithmicrequire}{\textbf{Input:}}
\renewcommand{\algorithmicensure}{\textbf{Output:}}
\algtext*{EndProcedure}\algtext*{EndIf}\algtext*{EndFor}
\begin{algorithmic}[1]
\Require Remaining sequence ($\mathcal K_{i,j}$), remaining timeout budget ($T_{i,j}$)
\Ensure \textsc{admit} or \textsc{skip}
\State $\mathcal D\gets\emptyset$ \label{alg:ad:init} \Comment{cleared when draft queue drains}
\State
\Procedure{DecideExecution}{$\mathcal K_{i,j},T_{i,j}$}
  \State \textbf{if} $g(k_j)\in\mathcal D$ \textbf{then} \Return \textsc{skip} \label{alg:ad:skip} \Comment{$g(k_j)\in\{M,S\}$}
  \State
  \State \textbf{/* Step 1. Estimation */}
  \State $\hat P\gets\mathit{predictor}.\Call{ComputePointEstimate}{\mathcal K_{i,j}}$ \label{alg:ad:estimate}
  \State $\phi_\tau\gets\textsf{SelectResidualFactor}(\mathcal K_{i,j})$ \label{alg:ad:select} \Comment{$\tau=n_{\mathrm{eff}}/(n_{\mathrm{eff}}+1)$}
  \State $U\gets\hat P\cdot\phi_\tau$ \label{alg:ad:upper}
  \State
  \State \textbf{/* Step 2. Decision */}
  \If{$U\le T_{i,j}$ \textbf{or} $\hat P=0$} \label{alg:ad:decide}
    \State \Return \textsc{admit}
  \Else
    \State $\mathcal D\gets\mathcal D\cup\{g(k_j)\}$ \label{alg:ad:disable}
    \State \Return \textsc{skip} \label{alg:ad:reject}
  \EndIf
\EndProcedure
\State
\Procedure{Update}{$\mathcal K_i$}
  \State \textbf{/* Step 3. Calibration */}
  \State $\phi\gets\emptyset$
  \For{$j\gets1$ \textbf{to} $m$}
    \State $(\mathcal K,\phi(\mathcal K))\gets\textsf{ComputeResidualFactor}(\mathcal K_{i,j})$ \label{alg:ad:residual}
  \EndFor
  \State $\textsf{UpdateResidualHistories}(\phi)$ \label{alg:ad:histories} \Comment{decay 0.90}
\EndProcedure
\end{algorithmic}
\end{algorithm}
