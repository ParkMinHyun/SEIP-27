\begin{algorithm}[t]
\caption{Remaining-Sequence Admission and Calibration}
\label{alg:admission}
\small
\renewcommand{\algorithmicrequire}{\textbf{Input:}}
\renewcommand{\algorithmicensure}{\textbf{Output:}}
\begin{algorithmic}[1]
\Require $\mathcal{K}_{i,j}$ and $R_{i,j}$ per decision; $p(k)$ per capture
\Ensure \textsc{skip} or \textsc{admit} with $W_{i,j}$; updated histories
\Procedure{DecideAdmission}{$\mathcal{K}_{i,j}$, $R_{i,j}$}
  \State $g \gets$ the optional stage's group \Comment{$M$ or $S$}
  \If{$g \in \mathit{demoted}$}
    \State \Return \textsc{skip}
  \EndIf
  \State record $\hat p(k)$ for $k \in \mathcal{K}_{i,j}$
  \If{$\hat P(\mathcal{K}_{i,j})>0$ \textbf{and} $U(\mathcal{K}_{i,j})>R_{i,j}$}
    \Comment{Eqs.~\ref{eq:upper-estimate},~\ref{eq:admission}}
    \State $\mathit{demoted} \gets \mathit{demoted} \cup \{g\}$
    \State \Return \textsc{skip}
  \EndIf
  \State $W_{i,j} \gets \max\{0,R_{i,j}-U(\mathcal{E}_i)\}$
    \Comment{Eq.~\ref{eq:watchdog}}
  \State \Return \textsc{admit} with watchdog $W_{i,j}$
\EndProcedure

\Statex
\Procedure{LearnFromCapture}{$i$} \Comment{at completion}
  \ForAll{measured parts priced in capture $i$}
    \State form $\phi(\mathcal{K})$ once per distinct part, if $\hat P>0$
      \Comment{Eq.~\ref{eq:residual-factor}}
  \EndFor
  \If{any factor was formed}
    \State $w_l \gets 0.90\,w_l$ for every stored $l$
    \State insert each into its history with unit weight
  \EndIf
  \State update $\hat b(k)$ and $\gamma$ \Comment{Section~\ref{sec:model}}
\EndProcedure
\end{algorithmic}
\end{algorithm}
