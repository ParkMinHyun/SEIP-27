\begin{algorithm}[t]
\caption{Capture Workflow with Admission and Pacing}
\label{alg:overview}
\footnotesize
\renewcommand{\algorithmicrequire}{\textbf{Input:}}
\renewcommand{\algorithmicensure}{\textbf{Output:}}
\algdef{SE}[EVENT]{Upon}{EndUpon}[2]{\textbf{upon} \textsc{#1}(#2) \textbf{do}}{\textbf{end upon}}
\algtext*{EndProcedure}\algtext*{EndUpon}\algtext*{EndIf}\algtext*{EndFor}
\begin{algorithmic}[1]
\Require User shutter input for capture $i$
\State $\mathit{predictor}\gets\textsf{SharedPredictor}()$
\State $\mathit{enabledGroups}\gets\{M,S\}$ \Comment{reset when the draft queue drains}
\State $\mathit{admission}\gets\textsf{Admission}(\mathit{predictor},\mathit{enabledGroups})$
\State $\mathit{pacing}\gets\textsf{Pacing}(\mathit{predictor},\mathit{enabledGroups})$
\State
\Procedure{Capture}{$i$}
  \State $\mathit{timeout}_i\gets\mathit{now}+\textsc{TIMEOUT}$ \label{alg:ov:timeout} \Comment{retained per capture}
  \State $\textsf{RequestCaptureBurst}(i)$
\EndProcedure
\State
\Upon{OnCaptureAvailable}{$i$}
  \State $\mathit{delay}_i\gets\mathit{pacing}.\Call{DecideDelay()}{}$ \label{alg:ov:pacing}
  \State $\textsf{NotifyCaptureAvailable}(i)$ after $\mathit{delay}_i$ \Comment{enables capture $i+1$}
\EndUpon
\State
\Upon{OnDraftFramesCollected}{$i,\mathit{frames}$}
  \State $\mathit{draft}_i\gets\textsf{CreateDraftSequence}(i,\mathit{frames})$
  \State $\textsf{Enqueue}(\mathit{draft}_i)$
\EndUpon
\State
\Upon{RunDraftSequence}{$\mathit{draft}_i$} \Comment{single worker, FIFO}
  \State $\mathcal K_i\gets\textsf{ConfigureStages}(\mathit{draft}_i)$ \label{alg:ov:configure}
  \For{each optional stage $k_j\in\mathcal K_i$}
    \State $T_{i,j}\gets\mathit{timeout}_i-\mathit{now}$
    \If{$\mathit{admission}.\Call{DecideExecution}{\mathcal K_{i,j},T_{i,j}}=\textsc{admit}$} \label{alg:ov:admission}
      \State $\textsf{Execute}(k_j)$
    \EndIf
  \EndFor
  \State $\textsf{PublishDraftImage}(\mathit{draft}_i)$ \Comment{encoding and saving} \label{alg:ov:publish}
\EndUpon
\State
\Upon{OnDraftSequenceCompleted}{$\mathcal K_i,C_i,p_i$}
  \State $\mathit{predictor}.\Call{ObserveDuration}{\mathcal K_i,C_i,p_i}$ \label{alg:ov:update}
  \State $\mathit{admission}.\Call{UpdateResidualHistories}{\mathcal K_i,p_i}$ \label{alg:ov:ad-residual}
  \State $\mathit{pacing}.\Call{RebaseBacklog}{\mathcal K_i}$ \label{alg:ov:pa-rebase}
\EndUpon
\end{algorithmic}
\end{algorithm}
