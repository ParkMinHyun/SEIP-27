% RQ3: Per-shot pacing behavior and run-level cost--tail trade-off for S26 Ultra.
%
% The figure currently renders the collected No-pacing and Ours series; empty
% policy CSVs add no curve until their factual traces arrive.
%
% Baseline naming follows the implementation
% (CaptureAvailableDelayPolicyFactory.PolicyType): NO_PACING, OURS,
% THERMAL_LUT, CODEL.  The earlier Queue-EWMA, "Static LUT", and "Queue
% dynamic" labels are retired.  The CSV stems stay no_pacing / ours /
% thermal_lut / codel_inspired.
%
% Each policy CSV contains one row per shot with realized backlog, queue depth,
% applied delay, pacing activation, and cumulative delay.  Panel (a) retains
% the median backlog, panel (b) shows the share of runs whose outgoing
% transition is paced, and panel (c) plots every policy's cumulative-delay
% median; only the two collected policies currently contribute IQR bands.
%
% Panel (d) complements rather than repeats the thresholded risk table.  Its
% x-axis is each run's total transition delay and its y-axis is that run's
% backlog P95.  One point represents one complete 30-shot run.  This
% continuous view remains informative when every paced policy has zero timeout
% or near-timeout exposure.
%
% The S26 Ultra series use the same included runs as
% tables/tab_rq3_pacing_summary.tex: No-pacing from
% 48U_metrics_12MP_normal_0729_AdmitOnly_{1,2}.xlsx (nine runs) and Ours from
% 48U_metrics_12MP_normal_0729.xlsx (eight runs), starting Lv3, 12MP normal.
% Cumulative cost at shot i sums delays on shots 1..i-1, because the delay
% recorded on shot j gates shot j+1.  It starts at zero and its shot-30 endpoint
% equals the full 29-transition run total.  Panel (d) uses the same 29 delays.
% The current figure remains provisional because the collected No-pacing and
% Ours runs did not match admitted workload.

  \providecommand{\rqThreeCumulativeDelayAxis}{ymin=-0.25}

  % delay, are drawn clear of the left axis line instead of on top of it.

  % Medians only; see the header comment for why the P10/P90 edges were dropped.

      table[x=shot,y=#2_median,col sep=comma]

      table[x=shot,y=#2_median,col sep=comma]
      {#1/ours.csv};
  }

  % Panel (b): percentage of complete runs with a positive transition delay.
  % Shot 30 is blank because its outgoing transition lies outside the window.
  \newcommand{\rqthreeactivationcurves}[1]{%
    \addplot[rq3 no pacing,forget plot]
      table[x=shot,y=activation_rate_percent,col sep=comma]
      {#1/no_pacing.csv};
    \addplot[rq3 thermal lut,forget plot]
      table[x=shot,y=activation_rate_percent,col sep=comma]

  }

  % Panel (c): cumulative cost is computed within runs before aggregation.
  % The IQR is shown only here because it directly communicates run-to-run

      table[x=shot,y=cumulative_delay_median,col sep=comma]

    \addplot[rq3 thermal lut,forget plot]
      table[x=shot,y=cumulative_delay_median,col sep=comma]
